    def ABGPSStepProbe(self):
        '''A/B harness: baro steps on a disarmed, still GPS copter'''
        self.set_parameters({"SIM_TERRAIN": 0})
        self.set_analog_rangefinder_parameters()
        self.reboot_sitl()
        self.wait_ready_to_arm()
        self.context_collect('STATUSTEXT')
        for step in [3, 5, 7, 10, -5]:
            self.context_clear_collection('STATUSTEXT')
            self.set_parameters({"SIM_BARO_GLITCH": step, "SIM_BAR2_GLITCH": step})
            tstart = self.get_sim_time_cached()
            polls = 0
            unhealthy = 0
            first_unhealthy = None
            maxvz = 0.0
            zjump = 0.0
            lastz = None
            while self.get_sim_time_cached() - tstart < 12:
                m = self.assert_receive_message('EKF_STATUS_REPORT', timeout=5)
                polls += 1
                if not (m.flags & mavutil.mavlink.ESTIMATOR_ATTITUDE):
                    unhealthy += 1
                    if first_unhealthy is None:
                        first_unhealthy = self.get_sim_time_cached() - tstart
                lp = self.assert_receive_message('LOCAL_POSITION_NED', timeout=5)
                maxvz = max(maxvz, abs(lp.vz))
                if lastz is not None:
                    zjump = max(zjump, abs(lp.z - lastz))
                lastz = lp.z
                self.send_mavlink_run_prearms_command()
                self.delay_sim_time(0.25, "poll")
            texts = sorted(set(x.text for x in self.context_collection('STATUSTEXT')
                               if "height error" in x.text or "settling" in x.text or "reset" in x.text.lower()))
            self.progress("ABGS step %d m: unhealthy %u/%u from %s s, max |vz| %.2f, max z step %.2f, final z %.2f, texts %s" % (
                step, unhealthy, polls, ('%.1f' % first_unhealthy) if first_unhealthy is not None else 'never',
                maxvz, zjump, lastz, texts[:5]))
            self.set_parameters({"SIM_BARO_GLITCH": 0, "SIM_BAR2_GLITCH": 0})
            self.delay_sim_time(20, "recover")
            self.wait_ready_to_arm(timeout=60)
